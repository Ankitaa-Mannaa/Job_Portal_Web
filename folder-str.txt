FOLDER STRUCTURE

BACKEND 

├── app/
│   ├── __init__.py                   # Flask app factory, JWT setup, blueprint registration
│   ├── config.py                     # Loads environment variables (SECRET_KEY, DB, API keys)
│   ├── db.py                         # MySQL DB connection helper
│   ├── wsgi.py                       # Entry point for running Flask
│
│   ├── auth/
│   │   ├── routes.py                 # /register and /login endpoints
│
│   ├── user/
│   │   ├── models.py                 # create_user(), get_user_by_email()
│
│   ├── roles/
│   │   ├── models.py                 # create_role(), get_role_by_name()
│
│   ├── resume/
│   │   ├── routes.py                 # Endpoint for uploading resumes
│   │   ├── models.py                 # upload_resume(), get_resume_by_user()
│   │   ├── parser.py                 # parse_resume(file_path)
│   │   ├── embeddings.py             # embed_text(text)
│   │   ├── vectordb.py               # ChromaDB logic for resume vector storage
│
│   ├── job/
│   │   ├── routes.py                 # CRUD APIs for job postings
│   │   ├── models.py                 # create_job(), get_job_by_id()
│
│   ├── application/
│   │   ├── routes.py                 # Apply to jobs, track status
│   │   ├── models.py                 # create_application(), get_applications_by_user()
│
│   ├── feedback/
│   │   ├── routes.py                 # HR adds feedback per interview
│   │   ├── models.py                 # add_feedback(), get_feedback_by_user()
│
│   ├── chatbot/
│   │   ├── routes.py                 # /chat endpoint with resume context
│   │   ├── ai_chatbot.py             # generate_chat(prompt, context)
│
│   └── routes.py                     # (optional) single file to import and register all blueprints


├── requirements.txt                  # Required packages
├── .env                              # Environment variables (not committed to repo)
└── README.md                         # Project setup instructions
